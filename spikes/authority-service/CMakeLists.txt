cmake_minimum_required(VERSION 3.20)
project(authority_service LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCO_GAME_SC OFF CACHE BOOL "" FORCE)
set(SCO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(NOT SDK_ROOT OR NOT EXISTS "${SDK_ROOT}/CMakeLists.txt" OR NOT EXISTS "${SDK_ROOT}/include/sco_net.h")
 message(FATAL_ERROR "SDK_ROOT must point to the checked, extracted core-source kit")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/VerifyExtractedKit.cmake")
sc_mp_verify_extracted_kit("${SDK_ROOT}" "908bd2674577dc435fce4bbf97c2cbbb75637ccba00de9375dafd3949cb09d0a" "authority service fixture")
add_subdirectory("${SDK_ROOT}" sdk EXCLUDE_FROM_ALL)
add_executable(authority-service main.cpp plugin.cpp)
target_link_libraries(authority-service PRIVATE sco_app)
target_compile_options(authority-service PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME authority-processes COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/service_process_tests.py" $<TARGET_FILE:authority-service>)
set_tests_properties(authority-processes PROPERTIES TIMEOUT 180)
add_executable(authority-service-fixture-tests fixture_tests.cpp plugin.cpp)
target_link_libraries(authority-service-fixture-tests PRIVATE sco_app)
target_compile_options(authority-service-fixture-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME authority-service-fixtures COMMAND authority-service-fixture-tests)
add_test(NAME authority-receipt-loss COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/receipt_loss_test.py" $<TARGET_FILE:authority-service>)
set_tests_properties(authority-service-fixtures PROPERTIES TIMEOUT 20)
set_tests_properties(authority-receipt-loss PROPERTIES TIMEOUT 30)
