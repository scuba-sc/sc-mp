cmake_minimum_required(VERSION 3.20)
project(authority_sdk_spike LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(SCO_GAME_SC OFF CACHE BOOL "" FORCE)
set(SCO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(NOT SDK_ROOT)
  message(FATAL_ERROR "Pass SDK_ROOT pointing to checksum-verified published core-source kit")
endif()
add_subdirectory("${SDK_ROOT}" sdk EXCLUDE_FROM_ALL)
add_executable(authority-cli main.cpp)
target_link_libraries(authority-cli PRIVATE sco_net)
target_include_directories(authority-cli PRIVATE "${SDK_ROOT}/tests")
if(SANITIZE)
  target_compile_options(authority-cli PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(authority-cli PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(Threads REQUIRED)
add_test(NAME cli-receipts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/cli_receipt_tests.py" $<TARGET_FILE:authority-cli>)
add_test(NAME authority COMMAND authority-cli test)
foreach(name receipt-tests receipt-integration medium-tests)
 string(REPLACE "-" "_" source "${name}")
 if(name STREQUAL "receipt-integration")
  set(source receipt_integration)
 endif()
 add_executable(${name} ${source}.cpp)
 target_link_libraries(${name} PRIVATE sco_net)
 if(name STREQUAL "receipt-tests")
  target_link_libraries(${name} PRIVATE Threads::Threads)
 endif()
 if(SANITIZE)
  target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${name} PRIVATE -fsanitize=address,undefined)
 endif()
 add_test(NAME ${name} COMMAND ${name})
endforeach()
