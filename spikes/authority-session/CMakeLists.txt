cmake_minimum_required(VERSION 3.20)
project(authority_session LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SCO_GAME_SC OFF CACHE BOOL "" FORCE)
set(SCO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(NOT SDK_ROOT OR NOT EXISTS "${SDK_ROOT}/CMakeLists.txt" OR NOT EXISTS "${SDK_ROOT}/include/sco/net/core.h")
 message(FATAL_ERROR "SDK_ROOT must point to the checked, extracted core-source kit")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/VerifyExtractedKit.cmake")
sc_mp_verify_extracted_kit("${SDK_ROOT}" "908bd2674577dc435fce4bbf97c2cbbb75637ccba00de9375dafd3949cb09d0a" "authority fixture")
add_subdirectory("${SDK_ROOT}" sdk EXCLUDE_FROM_ALL)
add_executable(authority-session main.cpp)
target_link_libraries(authority-session PRIVATE sco_net_service sco_net)
add_executable(authority-model-tests model_tests.cpp)
foreach(t authority-session authority-model-tests)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
add_test(NAME authority-model COMMAND authority-model-tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME authority-processes COMMAND "${Python3_EXECUTABLE}"
 "${CMAKE_CURRENT_SOURCE_DIR}/process_tests.py" $<TARGET_FILE:authority-session>)
set_tests_properties(authority-model PROPERTIES TIMEOUT 10)
set_tests_properties(authority-processes PROPERTIES TIMEOUT 90)
